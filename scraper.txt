javascript:(function(){
    // 1. Check if we are on Apple Music. If not, redirect there.
    if (!window.location.hostname.includes('music.apple.com')) {
        alert('Not on Apple Music. Redirecting you now...\n\nPlease click this bookmark again once the page fully loads.');
        window.location.href = 'https://music.apple.com';
        return;
    }

    // 2. Try to grab the tokens from MusicKit
    try {
        const music = MusicKit.getInstance();
        if (!music) {
            alert('MusicKit is not initialized yet. Please wait a second for the page to load and try again.');
            return;
        }

        const token = music.developerToken;
        const userToken = music.musicUserToken;

        if (!token || !userToken) {
            alert('Tokens missing! Please make sure you are signed into your Apple Music account in the browser.');
            return;
        }

        // 3. Send to Pipedream Webhook
        fetch('https://eos0gm5k9ykzpwo.m.pipedream.net', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ userToken: userToken, developerToken: token })
        })
        .then(response => {
            if (response.ok) {
                alert('Success! Apple Music tokens successfully sent and saved to GitHub!');
            } else {
                alert('Failed to send tokens. Server responded with status: ' + response.status);
            }
        })
        .catch(error => {
            alert('Network error while sending tokens: ' + error.message);
        });

    } catch (err) {
        alert('Error: ' + err.message + '\nMake sure you are logged into music.apple.com.');
    }
})();
